def get_season():
    month = datetime.datetime.now().month
    if month in [6,7,8,9,10]:   return "Kharif"
    elif month in [11,12,1,2,3]: return "Rabi"
    else:                         return "Zaid"


def get_top5_crops(model, encoder, N, P, K, temperature, humidity, ph, rainfall):
    sample = pd.DataFrame([[N, P, K, temperature, humidity, ph, rainfall]],
                          columns=['N','P','K','temperature','humidity','ph','rainfall'])
    proba = model.predict_proba(sample)[0]
    top5_idx   = np.argsort(proba)[::-1][:5]
    top5_crops = encoder.classes_[top5_idx]
    top5_conf  = proba[top5_idx] * 100
    return list(zip(top5_crops, top5_conf))


def get_fertilizer_recommendation(models, soil_type, top_crop, temperature, humidity, N, K, P):
    """Smart fertilizer recommendation with proper crop matching"""
    try:
        crop_fert_classes = list(models['crop_fert_encoder'].classes_)
        fert_crop = None
        # Exact match first
        for c in crop_fert_classes:
            if c.lower() == top_crop.lower():
                fert_crop = c
                break
        # Partial match
        if not fert_crop:
            for c in crop_fert_classes:
                if c.lower() in top_crop.lower() or top_crop.lower() in c.lower():
                    fert_crop = c
                    break
        # Map common crops to fertilizer dataset equivalents
        if not fert_crop:
            crop_map = {
                'rice': 'Paddy', 'banana': 'Sugarcane', 'mango': 'Sugarcane',
                'apple': 'Wheat', 'grapes': 'Cotton', 'orange': 'Sugarcane',
                'papaya': 'Sugarcane', 'coconut': 'Sugarcane',
                'chickpea': 'Pulses', 'lentil': 'Pulses',
                'kidneybeans': 'Pulses', 'blackgram': 'Pulses',
                'mungbean': 'Pulses', 'mothbeans': 'Pulses',
                'pigeonpeas': 'Pulses', 'pomegranate': 'Cotton',
                'watermelon': 'Millets', 'muskmelon': 'Millets',
                'jute': 'Cotton', 'coffee': 'Sugarcane',
            }
            mapped = crop_map.get(top_crop.lower())
            if mapped and mapped in crop_fert_classes:
                fert_crop = mapped

        if not fert_crop:
            fert_crop = crop_fert_classes[0]

        soil_enc = models['soil_encoder'].transform([soil_type])[0]
        crop_enc = models['crop_fert_encoder'].transform([fert_crop])[0]
        sample = pd.DataFrame([[temperature, humidity, 40, soil_enc, crop_enc, N, K, P]],
                              columns=['Temparature','Humidity','Moisture',
                                       'soil_encoded','crop_encoded',
                                       'Nitrogen','Potassium','Phosphorous'])
        pred = models['fertilizer_model'].predict(sample)[0]
        return models['fertilizer_encoder'].inverse_transform([pred])[0]
    except Exception:
        return "Balanced NPK"

def geocode_location(location):
    api_key = st.secrets.get("WEATHER_API_KEY", "")
    if not api_key:
        return None, "WEATHER_API_KEY not set in secrets.toml"
    try:
        r = requests.get("http://api.openweathermap.org/geo/1.0/direct",
                         params={"q": location, "limit": 5, "appid": api_key}, timeout=5)
        data = r.json()
        if r.status_code != 200 or not data:
            return None, "No locations found. Try a different spelling."
        return data, None
    except Exception as e:
        return None, str(e)


def fetch_weather_by_coords(lat, lon):
    import datetime as _dt
    api_key = st.secrets.get("WEATHER_API_KEY", "")
    try:
        r = requests.get("https://api.openweathermap.org/data/2.5/weather",
                         params={"lat": lat, "lon": lon, "appid": api_key, "units": "metric"}, timeout=5)
        wx = r.json()
        temperature = wx["main"]["temp"]
        humidity    = wx["main"]["humidity"]
        today        = _dt.date.today()
        one_year_ago = today - _dt.timedelta(days=30)
        rr = requests.get("https://archive-api.open-meteo.com/v1/archive",
                          params={"latitude": lat, "longitude": lon,
                                  "start_date": str(one_year_ago), "end_date": str(today),
                                  "daily": "precipitation_sum", "timezone": "auto"}, timeout=15)
        rain_data = rr.json()
        annual_rainfall = round(sum(v for v in rain_data["daily"]["precipitation_sum"] if v is not None), 1)
        raw_rainfall    = annual_rainfall
        annual_rainfall = min(annual_rainfall, 300.0)
        return temperature, humidity, annual_rainfall, raw_rainfall, None
    except Exception as e:
        return None, None, None, None, str(e)

def parse_advice_sections(advice_text):
    section_map = {
        "SEED SELECTION":            "🌾",
        "FERTILIZER SCHEDULE":       "💊",
        "IRRIGATION PLAN":           "💧",
        "DISEASE":                   "🦠",
        "EXPECTED CHALLENGES":       "⚠️",
        "HARVEST GUIDANCE":          "🌿",
    }
    display_names = {
        "SEED SELECTION":      "Seed Selection",
        "FERTILIZER SCHEDULE": "Fertilizer Schedule",
        "IRRIGATION PLAN":     "Irrigation Plan",
        "DISEASE":             "Disease & Pest Prevention",
        "EXPECTED CHALLENGES": "Expected Challenges",
        "HARVEST GUIDANCE":    "Harvest Guidance",
    }
    result = {}
    current_section = None
    current_lines   = []
    for line in advice_text.split('\n'):
        line_clean = line.strip().upper().replace('#','').replace('*','').strip()
        matched = False
        for section in section_map:
            if section in line_clean:
                if current_section:
                    result[current_section] = '\n'.join(current_lines).strip()
                current_section = section
                current_lines   = []
                matched = True
                break
        if not matched and current_section:
            current_lines.append(line)
    if current_section:
        result[current_section] = '\n'.join(current_lines).strip()
    return result, section_map, display_names

def show_yield_gap_analysis(top_crop, pred, low, high, N, P, K, ph, shap_pairs):
    """Feature 5: Yield Gap Analysis"""
    st.markdown('<div class="section-title">📊 Yield Gap Analysis</div>', unsafe_allow_html=True)

    try:
        yield_df = pd.read_csv('datasets/cleaned_yield.csv')
        crop_data = yield_df[yield_df['crop'].str.lower() == top_crop.lower()]

        if len(crop_data) == 0:
            st.info("Yield gap analysis not available for this crop in our dataset.")
            return

        avg_yield  = round(crop_data['yield_tonnes_per_ha'].mean(), 2)
        best_yield = round(crop_data['yield_tonnes_per_ha'].quantile(0.90), 2)
        gap        = round(best_yield - pred, 2)
        gap_pct    = round((gap / best_yield) * 100, 1) if best_yield > 0 else 0

        # Visual comparison bar
        max_val = max(best_yield * 1.1, pred * 1.1)

        your_pct  = int((pred / max_val) * 100)
        avg_pct   = int((avg_yield / max_val) * 100)
        best_pct  = int((best_yield / max_val) * 100)

        st.markdown(
            f'<div class="feature-card">'
            f'<div class="feature-card-title">📊 Yield Comparison</div>'
            f'<div style="margin:0.6rem 0">'
            f'<div style="display:flex;align-items:center;gap:0.8rem;margin:0.5rem 0">'
            f'<span style="width:120px;font-size:0.82rem;font-weight:700;color:#1b5e20">Your Yield</span>'
            f'<div style="flex:1;background:#e8f5e9;border-radius:6px;height:16px;overflow:hidden">'
            f'<div style="width:{your_pct}%;height:16px;background:linear-gradient(90deg,#1b5e20,#ffd700);border-radius:6px"></div></div>'
            f'<span style="width:80px;font-size:0.88rem;font-weight:800;color:#1b5e20">{pred:.1f} t/ha</span>'
            f'</div>'
            f'<div style="display:flex;align-items:center;gap:0.8rem;margin:0.5rem 0">'
            f'<span style="width:120px;font-size:0.82rem;font-weight:700;color:#f57c00">Global Average</span>'
            f'<div style="flex:1;background:#fff3e0;border-radius:6px;height:16px;overflow:hidden">'
            f'<div style="width:{avg_pct}%;height:16px;background:#f57c00;border-radius:6px"></div></div>'
            f'<span style="width:80px;font-size:0.88rem;font-weight:800;color:#f57c00">{avg_yield:.1f} t/ha</span>'
            f'</div>'
            f'<div style="display:flex;align-items:center;gap:0.8rem;margin:0.5rem 0">'
            f'<span style="width:120px;font-size:0.82rem;font-weight:700;color:#c62828">Top 10% Yield</span>'
            f'<div style="flex:1;background:#ffebee;border-radius:6px;height:16px;overflow:hidden">'
            f'<div style="width:{best_pct}%;height:16px;background:#c62828;border-radius:6px"></div></div>'
            f'<span style="width:80px;font-size:0.88rem;font-weight:800;color:#c62828">{best_yield:.1f} t/ha</span>'
            f'</div>'
            f'</div>'
            f'<div style="background:linear-gradient(135deg,rgba(198,40,40,0.08),rgba(255,255,255,0.9));'
            f'border-radius:10px;padding:0.8rem;border-left:4px solid #c62828;margin-top:0.8rem">'
            f'<b>Yield Gap: {gap:.1f} t/ha ({gap_pct}% below top performers)</b><br>'
            f'<span style="font-size:0.82rem;color:#6b7280">Closing this gap could increase your income significantly.</span>'
            f'</div>'
            f'</div>',
            unsafe_allow_html=True
        )

        # What's causing the gap — from SHAP
        if shap_pairs:
            st.markdown("**🔍 What's causing your yield gap:**")
            neg_factors = [(f, v) for f, v in shap_pairs if v < 0]
            if neg_factors:
                for feat, val in neg_factors[:3]:
                    # Map feature to improvement advice
                    advice_map = {
                        'Nitrogen':    f'Your N={N} kg/ha. Increasing to 100-120 kg/ha could reduce gap by ~0.5-1.0 t/ha',
                        'Phosphorus':  f'Your P={P} kg/ha. Optimal P improves root system and yield significantly',
                        'Potassium':   f'Your K={K} kg/ha. K improves grain filling and disease resistance',
                        'pH':          f'Your pH={ph}. Correcting pH unlocks existing nutrients in soil',
                        'Rainfall':    f'Rainfall of {N}mm is limiting. Consider irrigation to supplement rainfall',
                        'Temperature': f'Temperature optimization through sowing date adjustment may help',
                        'Humidity':    f'Humidity conditions are slightly unfavorable — monitor for disease',
                    }
                    improvement = advice_map.get(feat, f'{feat} is limiting yield potential')
                    st.markdown(
                        f'<div class="soil-fix-step" style="border-left-color:#c62828">'
                        f'🔴 <b>{feat}</b> is limiting your yield<br>'
                        f'<span style="font-size:0.82rem;color:#6b7280">{improvement}</span>'
                        f'</div>',
                        unsafe_allow_html=True
                    )
            else:
                st.success("✅ Your soil and climate conditions are well-optimized. Yield gap is mainly due to regional factors.")

    except Exception:
        st.info("Yield gap analysis not available for this crop/region.")
